using Wulato.UIKit;
public sealed class ArgsView : UIView<int> { }
public static class ContractCompileProbe
{
    public static void Check(IUIContext ui)
    {
#if UIKIT_NEGATIVE_PROBE
        ui.OpenAsync<ArgsView>();
#else
        ui.OpenAsync<ArgsView,int>(42);
#endif
    }
}

#if UIKIT_NAVIGATION_PROBE
public static class NavigationCompileProbe
{
    public static void Check(Wulato.UIKit.Navigation.IUINavigator navigator)
    {
#if UIKIT_NEGATIVE_PROBE
        navigator.PushAsync<ArgsView>();
#else
        navigator.PushAsync<ArgsView,int>(42);
#endif
    }
}
#endif
